    <script>
        // 字符集定义
        const charSets = {
            26: 'abcdefghijklmnopqrstuvwxyz',
            32: 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789', // 不包含 ILOU
            36: '0123456789abcdefghijklmnopqrstuvwxyz',
            52: 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ',
            58: '123456789ABCDEFGHJKLMNPQRSTUVWXYZabcdefghijkmnopqrstuvwxyz', // 不包含 0OIl
            62: '0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ'
        };
        
        // 初始化
        function init() {
            // 绑定事件
            document.getElementById('customBase').addEventListener('change', function() {
                const base = this.value;
                document.querySelectorAll('input[name="inputBase"]').forEach(radio => {
                    radio.checked = radio.value === base;
                });
            });
            
            // 初始转换
            convert();
        }
        
        // 转换函数
        function convert() {
            const inputValue = document.getElementById('inputValue').value.trim();
            if (!inputValue) {
                alert('请输入数值');
                return;
            }
            
            // 获取输入进制
            const inputBase = parseInt(document.querySelector('input[name="inputBase"]:checked').value);
            
            // 转换为十进制
            let decimalValue;
            try {
                decimalValue = parseInt(inputValue, inputBase);
                if (isNaN(decimalValue)) {
                    throw new Error('无效的输入值');
                }
            } catch (error) {
                alert('输入值与所选进制不匹配');
                return;
            }
            
            // 转换到各个进制
            convertToBase(decimalValue, 2, 'result2');
            convertToBase(decimalValue, 8, 'result8');
            convertToBase(decimalValue, 10, 'result10');
            convertToBase(decimalValue, 16, 'result16');
            convertToBase(decimalValue, 26, 'result26', charSets[26]);
            convertToBase(decimalValue, 32, 'result32', charSets[32]);
            convertToBase(decimalValue, 36, 'result36', charSets[36]);
            convertToBase(decimalValue, 52, 'result52', charSets[52]);
            convertToBase(decimalValue, 58, 'result58', charSets[58]);
            convertToBase(decimalValue, 62, 'result62', charSets[62]);
        }
        
        // 转换到指定进制
        function convertToBase(value, base, resultId, charSet = null) {
            let result;
            
            if (charSet) {
                // 使用自定义字符集
                result = convertWithCharSet(value, charSet);
            } else {
                // 使用内置转换
                result = value.toString(base);
                // 16进制大写
                if (base === 16) {
                    result = result.toUpperCase();
                }
            }
            
            // 二进制特殊处理
            if (base === 2) {
                const padZero = document.getElementById('padZero').checked;
                if (padZero) {
                    const padBits = parseInt(document.getElementById('padBits').value);
                    const endian = document.querySelector('input[name="endian"]:checked').value;
                    
                    // 补零
                    while (result.length < padBits) {
                        result = '0' + result;
                    }
                    
                    // 小端序处理
                    if (endian === 'little') {
                        result = reverseBits(result, 8); // 按字节反转
                    }
                }
            }
            
            // 显示结果
            document.getElementById(resultId).value = result;
        }
        
        // 使用自定义字符集转换
        function convertWithCharSet(value, charSet) {
            if (value === 0) return charSet[0];
            
            const base = charSet.length;
            let result = '';
            let num = value;
            
            while (num > 0) {
                const remainder = num % base;
                result = charSet[remainder] + result;
                num = Math.floor(num / base);
            }
            
            return result;
        }
        
        // 按字节反转二进制字符串
        function reverseBits(binString, byteSize) {
            // 确保长度是字节的倍数
            while (binString.length % byteSize !== 0) {
                binString = '0' + binString;
            }
            
            // 按字节分割
            const bytes = [];
            for (let i = 0; i < binString.length; i += byteSize) {
                bytes.push(binString.substring(i, i + byteSize));
            }
            
            // 反转字节顺序
            bytes.reverse();
            
            // 重新组合
            return bytes.join('');
        }
        
        // 初始化
        init();
    </script>
